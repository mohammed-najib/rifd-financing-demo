import assert from 'node:assert/strict'
import { mkdir, writeFile } from 'node:fs/promises'
import webdriver from '/home/mdnb/.npm/_npx/deb77af66d9637d4/node_modules/selenium-webdriver/index.js'
import firefox from '/home/mdnb/.npm/_npx/deb77af66d9637d4/node_modules/selenium-webdriver/firefox.js'
const [origin, directory] = process.argv.slice(2)
await mkdir(directory, { recursive: true })
const options = new firefox.Options().setBinary('/usr/bin/firefox').addArguments('-headless', '--no-remote')
const service = new firefox.ServiceBuilder('/home/mdnb/.cache/selenium/geckodriver/linux64/0.37.1/geckodriver')
const driver = await new webdriver.Builder().forBrowser('firefox').setFirefoxOptions(options).setFirefoxService(service).build()
const captures = []
const wait = fn => driver.wait(() => driver.executeScript(fn), 30000)
const visit = async path => { await driver.get(origin + path); await wait(() => !!document.querySelector('[data-workspace-ready="true"]')) }
const click = text => driver.findElement(webdriver.By.xpath(`//button[@aria-label='${text}' or normalize-space(.)='${text}']`)).click()
async function capture(name, language) {
 const data = await driver.executeScript(() => ({ url: location.href, language: document.documentElement.lang, direction: document.documentElement.dir, heading: document.querySelector('main h1')?.textContent, identity: document.querySelector('.identity-name')?.textContent, width: innerWidth, fonts: [...document.fonts].map(f => ({family:f.family, status:f.status})), scripts: [...document.scripts].map(s=>s.src).filter(Boolean) }))
 assert.equal(data.language, language)
 assert.equal(data.direction, language === 'ar' ? 'rtl' : 'ltr')
 assert(data.fonts.filter(f => /Noto/.test(f.family)).every(f => f.status === 'loaded'))
 await writeFile(`${directory}/${name}.png`, await driver.takeScreenshot(), 'base64')
 captures.push({ name, ...data })
 console.log(name)
}
try {
 const capabilities = await driver.getCapabilities()
 for (const language of ['ar', 'en']) {
  const ar = language === 'ar'
  await visit('/products')
  if (!ar) { await click('English'); await wait(() => document.documentElement.lang === 'en') }
  await capture(`${language}-products`, language)
  await visit('/applications/RIFD-2026-001/edit?step=business')
  assert.equal(await driver.findElements(webdriver.By.id('field-businessName')).then(x=>x.length), 0)
  await click(ar ? 'مريم سالم' : 'Mariam Salem')
  await wait(() => !!document.querySelector('#field-businessName'))
  assert.equal(await driver.findElement(webdriver.By.id('field-businessName')).getAttribute('value'), 'متجر السهل التجريبي')
  await driver.navigate().refresh()
  await wait(() => !!document.querySelector('#field-businessName'))
  await capture(`${language}-applicant-refresh`, language)
  await visit('/applications/RIFD-2026-005/repayment')
  await wait(() => !!document.querySelector('#repayment-totals'))
  await capture(`${language}-repayment`, language)
  await click(ar ? 'تغيير الهوية' : 'Switch identity')
  await click(ar ? 'سلمى ناصر' : 'Salma Nasser')
  await wait(() => !!document.querySelector('#queue-sort'))
  await visit('/staff/applications/RIFD-2026-007')
  await wait(() => !!document.querySelector('#review-information'))
  await driver.navigate().refresh()
  await wait(() => !!document.querySelector('#review-information'))
  await capture(`${language}-staff-refresh`, language)
  await visit('/staff/applications/RIFD-2026-005/print')
  await wait(() => !!document.querySelector('.print-summary'))
  await capture(`${language}-print`, language)
  await click(ar ? 'تسجيل الخروج' : 'Sign out')
  await visit('/demo/reset')
  await capture(`${language}-reset-warning`, language)
 }
 await writeFile(`${directory}/result.json`, JSON.stringify({ outcome:'passed', checkedAt:new Date().toISOString(), origin, browser:capabilities.get('browserName'), version:capabilities.get('browserVersion'), headless:true, scope:'Fresh startup, public products, signed-out protected link, identity return, business query and refresh, approved repayment, reviewer role switch and refresh, public print and reset warning in both languages. Full journeys use the separate Playwright reports.', captures }, null, 2)+'\n')
} finally { await driver.quit() }
